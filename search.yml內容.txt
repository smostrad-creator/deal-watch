name: 手機查機票

on:
  workflow_dispatch:
    inputs:
      params:
        description: "查詢條件（由查詢頁自動產生，不用自己填）"
        required: true

permissions:
  contents: write

concurrency:
  group: search
  cancel-in-progress: false

jobs:
  search:
    runs-on: ubuntu-latest
    timeout-minutes: 60
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: 安裝工具
        run: pip install faster-flights openpyxl

      - name: 查機票
        env:
          SEARCH_PARAMS: ${{ inputs.params }}
          TELEGRAM_TOKEN: ${{ secrets.TELEGRAM_TOKEN }}
          TELEGRAM_CHAT_ID: ${{ secrets.TELEGRAM_CHAT_ID }}
          SERPAPI_KEY: ${{ secrets.SERPAPI_KEY }}
          SERPAPI_MAX: "40"
          PAGES_URL: https://${{ github.repository_owner }}.github.io/${{ github.event.repository.name }}/
        run: python search_job.py

      - name: 存結果給查詢頁
        run: |
          git config user.name "flight-search-bot"
          git config user.email "flight-search-bot@users.noreply.github.com"
          git add docs/results
          git diff --cached --quiet && exit 0
          git commit -m "查詢結果"
          for i in 1 2 3; do
            git pull --rebase origin main && git push && break
            sleep 5
          done
